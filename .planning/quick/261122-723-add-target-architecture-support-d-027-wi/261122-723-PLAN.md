```yaml
      - name: Install rustup if missing
        if: runner.os == 'Windows'
        shell: pwsh
        run: |
          if (Get-Command rustup -ErrorAction SilentlyContinue) {
            Write-Host "rustup preinstalled: yes"
          } else {
            Write-Host "rustup preinstalled: no"
            $triple = if ($env:RUNNER_ARCH -eq 'ARM64') { 'aarch64-pc-windows-msvc' } else { 'x86_64-pc-windows-msvc' }
            $installer = Join-Path $env:RUNNER_TEMP 'rustup-init.exe'
            Invoke-WebRequest -Uri "https://static.rust-lang.org/rustup/dist/$triple/rustup-init.exe" -OutFile $installer
            & $installer -y --default-toolchain none --profile minimal
            if ($LASTEXITCODE -ne 0) { exit $LASTEXITCODE }
            Add-Content -Path $env:GITHUB_PATH -Value "$env:USERPROFILE\.cargo\bin"
          }
      - name: Install rustup if missing
        if: runner.os != 'Windows'
        shell: bash
        run: |
          if command -v rustup >/dev/null 2>&1; then
            echo "rustup preinstalled: yes"
          else
            echo "rustup preinstalled: no"
            curl --proto '=https' --tlsv1.2 -sSf https://sh.rustup.rs | sh -s -- -y --default-toolchain none --profile minimal
            echo "$HOME/.cargo/bin" >> "$GITHUB_PATH"
          fi
```
